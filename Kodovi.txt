CREATE TABLE OSOBA(
    OSOBA_ID INT GENERATED ALWAYS AS IDENTITY,
    IME VARCHAR2(25) NOT NULL,
    PREZIME VARCHAR2(30) NOT NULL,
    JMBG CHAR(13) NOT NULL,
    ADRESA VARCHAR2(100),
    POL CHAR NOT NULL,
    DATUM_RODJENJA DATE NOT NULL,
    EMAIL VARCHAR(100) NOT NULL,
    KOMENTAR VARCHAR(100),
    
    CONSTRAINT PK_OSOBA PRIMARY KEY (OSOBA_ID),
    CONSTRAINT UQ_JMBG UNIQUE (JMBG),
    CONSTRAINT CK_JMBG_VALIDITY CHECK (REGEXP_LIKE(JMBG, '^[0-9]{13}$')),
    CONSTRAINT CK_POL CHECK(POL IN('M', 'Z', 'Ž')),
    CONSTRAINT CK_EMAIL_VALIDITY CHECK (REGEXP_LIKE(email, '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'))
)

CREATE TABLE RODITELJ_STARATELJ(
    ID_OSOBA INT NOT NULL,
    ZANIMANJE VARCHAR(50) NOT NULL,
    RADNO_MESTO VARCHAR(50),
    CONSTRAINT FK_OSOBA FOREIGN KEY(ID_OSOBA) REFERENCES OSOBA(OSOBA_ID)
)


CREATE TABLE NASTAVNIK(
    ID_OSOBA INT NOT NULL,
    STRUCNA_SPREMA VARCHAR(20) NOT NULL,
    STATUS VARCHAR(20) NOT NULL,
    ZVANJE VARCHAR(50) NOT NULL,
    DATUM_ZAPOSLENJA DATE NOT NULL,
    
    CONSTRAINT FK_NASTAVNIK_OSOBA FOREIGN KEY (ID_OSOBA) REFERENCES OSOBA(OSOBA_ID),
    CONSTRAINT CK_NASTAVNIK_STATUS CHECK (STATUS IN ('AKTIVAN', 'ODSUTAN', 'NA BOLOVANJU', 'PENZIONISAN'))
)


CREATE TABLE ODELJENJE(
    OZNAKA VARCHAR(10) NOT NULL,
    SKOLSKA_GODINA VARCHAR(10) NOT NULL,
    RAZRED NUMBER NOT NULL,
    
    CONSTRAINT PK_ODELJENJE PRIMARY KEY (RAZRED, OZNAKA, SKOLSKA_GODINA),
    CONSTRAINT CK_ODELJENJE_RAZRED CHECK (RAZRED BETWEEN 1 AND 8),
    CONSTRAINT CK_ODELJENJE_GODINA CHECK (REGEXP_LIKE(SKOLSKA_GODINA, '^[1-9][0-9]{3}/[1-9][0-9]{3}$'))
);

CREATE TABLE UCENIK(
    RAZRED NUMBER NOT NULL,
    STATUS VARCHAR2(10) NOT NULL,
    GODINA_UPISA VARCHAR2(10) NOT NULL,
    CONSTRAINT CK_UCENIK_RAZRED CHECK (RAZRED BETWEEN 1 AND 8),
    CONSTRAINT CK_GODINA_UPISA CHECK (REGEXP_LIKE(GODINA_UPISA, '^[1-9][0-9]{3}/[1-9][0-9]{3}$'))
)
