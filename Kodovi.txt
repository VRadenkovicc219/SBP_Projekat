CREATE TABLE OSOBA(
    ID INT GENERATED ALWAYS AS IDENTITY,
    IME VARCHAR2(25) NOT NULL,
    PREZIME VARCHAR2(30) NOT NULL,
    JMBG CHAR(13) NOT NULL,
    ADRESA VARCHAR2(100),
    POL CHAR(1 CHAR) NOT NULL,
    DATUM_RODJENJA DATE NOT NULL,
    EMAIL VARCHAR2(100) NOT NULL,
    KOMENTAR VARCHAR2(100),
    
    CONSTRAINT PK_OSOBA PRIMARY KEY (ID),
    CONSTRAINT UQ_OSOBA_JMBG UNIQUE (JMBG),
    CONSTRAINT UQ_OSOBA_EMAIL UNIQUE (EMAIL),
    CONSTRAINT CK_OSOBA_JMBG CHECK (REGEXP_LIKE(JMBG, '^[0-9]{13}$')),
    CONSTRAINT CK_OSOBA_POL CHECK (POL IN ('M', 'Z', 'Ž')),
    CONSTRAINT CK_OSOBA_EMAIL CHECK (REGEXP_LIKE(EMAIL, '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'))
);
	
CREATE TABLE RODITELJ_STARATELJ(
    ID INT GENERATED ALWAYS AS IDENTITY,
    ID_OSOBA INT NOT NULL,
    ZANIMANJE VARCHAR2(50) NOT NULL,
    RADNO_MESTO VARCHAR2(50),
    
    CONSTRAINT PK_RODITELJ PRIMARY KEY (ID),
    CONSTRAINT FK_RODITELJ_OSOBA FOREIGN KEY (ID_OSOBA) REFERENCES OSOBA(ID)
);

CREATE TABLE NASTAVNIK(
    ID INT GENERATED ALWAYS AS IDENTITY,
    ID_OSOBA INT NOT NULL,
    STRUCNA_SPREMA VARCHAR2(20) NOT NULL,
    STATUS VARCHAR2(20) NOT NULL,
    ZVANJE VARCHAR2(50) NOT NULL,
    DATUM_ZAPOSLENJA DATE NOT NULL,
    
    CONSTRAINT PK_NASTAVNIK PRIMARY KEY (ID),
    CONSTRAINT FK_NASTAVNIK_OSOBA FOREIGN KEY (ID_OSOBA) REFERENCES OSOBA(ID),
    CONSTRAINT CK_NASTAVNIK_STATUS CHECK (STATUS IN ('AKTIVAN', 'ODSUTAN', 'NA BOLOVANJU', 'PENZIONISAN'))
);

CREATE TABLE ODELJENJE(
    ID INT GENERATED ALWAYS AS IDENTITY,
    OZNAKA VARCHAR2(10) NOT NULL,
    SKOLSKA_GODINA VARCHAR2(10) NOT NULL,
    RAZRED NUMBER NOT NULL,
    
    CONSTRAINT PK_ODELJENJE PRIMARY KEY (ID),
    CONSTRAINT UQ_ODELJENJE UNIQUE (RAZRED, OZNAKA, SKOLSKA_GODINA),
    CONSTRAINT CK_ODELJENJE_RAZRED CHECK (RAZRED BETWEEN 1 AND 8),
    CONSTRAINT CK_ODELJENJE_GODINA CHECK (REGEXP_LIKE(SKOLSKA_GODINA, '^[1-9][0-9]{3}/[1-9][0-9]{3}$'))
);

CREATE TABLE UCENIK(
    ID INT GENERATED ALWAYS AS IDENTITY,
    ID_OSOBA INT NOT NULL,
    STATUS VARCHAR2(10) NOT NULL,
    GODINA_UPISA VARCHAR2(10) NOT NULL,
    
    CONSTRAINT PK_UCENIK PRIMARY KEY (ID),
    CONSTRAINT FK_UCENIK_OSOBA FOREIGN KEY (ID_OSOBA) REFERENCES OSOBA(ID),
    CONSTRAINT CK_UCENIK_GODINA_UPISA CHECK (REGEXP_LIKE(GODINA_UPISA, '^[1-9][0-9]{3}/[1-9][0-9]{3}$'))
);

CREATE TABLE STARATELJSTVO(
    ID INT GENERATED ALWAYS AS IDENTITY,
    ID_UCENIK INT NOT NULL,
    ID_STARATELJ INT NOT NULL,
    
    CONSTRAINT PK_STARATELJSTVO PRIMARY KEY(ID),

    CONSTRAINT FK_STARATELJSTVO_STARATELJ FOREIGN KEY(ID_STARATELJ) REFERENCES RODITELJ_STARATELJ(ID),
    CONSTRAINT UQ_STARATELJ_UCENIIK UNIQUE(ID_UCENIK, ID_STARATELJ)
);

CREATE TABLE UCENIK_ODELJENJE(
    ID INT GENERATED ALWAYS AS IDENTITY,

    CONSTRAINT FK_ODELJENJE_ODELJENJE FOREIGN KEY(ID_UCENIK) REFERENCES ODELJENJE(ID),
    CONSTRAINT UQ_UCENIK_ODELJENJE UNIQUE(ID_UCENIK, ID_ODELJENJE)

CREATE TABLE PREDMET(
    ID INT GENERATED ALWAYS AS IDENTITY,
    NAZIV VARCHAR2(50) NOT NULL,
    SKOLSKA_GODINA VARCHAR2(10) NOT NULL,
    RAZRED NUMBER(1) NOT NULL,
    NEDELJNI_FOND NUMBER NOT NULL,
    TIP VARCHAR2(12) NOT NULL,
    OPIS VARCHAR2(50),
    KOMENTAR VARCHAR2(50)
    
    CONSTRAINT PK_PREDMET PRIMARY KEY(ID),
    CONSTRAINT CK_RAZRED_PREDMET CHECK(RAZRED BETWEEN 1 AND 8),

CREATE TABLE SLUSA_PREDMET(

CREATE OR REPLACE TRIGGER TRG_DISJOINT_RAZREDNI
BEFORE INSERT OR UPDATE ON RAZREDNI_STARESINA
FOR EACH ROW
DECLARE
  V_COUNT NUMBER;
BEGIN
  SELECT COUNT(*) INTO V_COUNT FROM (
    SELECT ID_NASTAVNIK FROM RUKOVODECE_OSOBLJE WHERE ID_NASTAVNIK = :NEW.ID_NASTAVNIK
    UNION ALL
    SELECT ID_NASTAVNIK FROM STRUCNI_SARADNIK WHERE ID_NASTAVNIK = :NEW.ID_NASTAVNIK
  );
  IF V_COUNT > 0 THEN
    RAISE_APPLICATION_ERROR(-20012, 'Nastavnik moze imati najvise jednu dodatnu ulogu');
  END IF;
END;
/
ALTER TRIGGER TRG_DISJOINT_RAZREDNI ENABLE;
